type Script struct {
	src       string
	mu        sync.RWMutex
	hash      string
	serverSHA bool // if true: do not compute SHA-1 in Go; load digest from Redis (SCRIPT LOAD)
}

func NewScript(src string) *Script {
	h := sha1.New()
	_, _ = io.WriteString(h, src)

	return &Script{
		src:       src,
		hash:      hex.EncodeToString(h.Sum(nil)),
		serverSHA: false,
	}
}

// NewScriptServerSHA creates a Script that avoids computing SHA-1 in Go.
// The digest is obtained from Redis via SCRIPT LOAD (server-side hashing),
// then EVALSHA/EVALSHA_RO is used.
func NewScriptServerSHA(src string) *Script {
	return &Script{
		src:       src,
		serverSHA: true,
	}
}

func (s *Script) Hash() string {
	s.mu.RLock()
	defer s.mu.RUnlock()
	return s.hash
}

func (s *Script) Load(ctx context.Context, c Scripter) *StringCmd {
	cmd := c.ScriptLoad(ctx, s.src)
	if err := cmd.Err(); err == nil {
		s.mu.Lock()
		s.hash = cmd.Val()
		s.mu.Unlock()
	}
	return cmd
}

func (s *Script) Exists(ctx context.Context, c Scripter) *BoolSliceCmd {
	s.mu.RLock()
	hash := s.hash
	serverSHA := s.serverSHA
	s.mu.RUnlock()
	if hash == "" && serverSHA {
		// For server-side scripts, obtain digest from Redis first.
		// If hash is empty, it means SCRIPT LOAD was not called yet, so we check existence of empty hash which will return false.
		// This avoids unnecessary SCRIPT LOAD just to check existence.
		if err := s.ensureHash(ctx, c); err != nil {
			return c.ScriptExists(ctx, "")
		}
		s.mu.RLock()
		hash = s.hash
		s.mu.RUnlock()
	}
	if hash == "" {
		return c.ScriptExists(ctx, "")
	}
	return c.ScriptExists(ctx, hash)
}

func (s *Script) Eval(ctx context.Context, c Scripter, keys []string, args ...interface{}) *Cmd {
	return c.Eval(ctx, s.src, keys, args...)
}

func (s *Script) EvalRO(ctx context.Context, c Scripter, keys []string, args ...interface{}) *Cmd {
	return c.EvalRO(ctx, s.src, keys, args...)
}

// ensureHash ensures that s.hash is populated by using SCRIPT LOAD.
// It never calls SHA-1 in Go; Redis computes and returns the digest.
func (s *Script) ensureHash(ctx context.Context, c Scripter) error {
	// Fast path: read lock, return if hash is already set.
	s.mu.RLock()
	if s.hash != "" {
		s.mu.RUnlock()
		return nil
	}
	s.mu.RUnlock()

	// Slow path: acquire write lock and load.
	s.mu.Lock()
	if s.hash != "" {
		s.mu.Unlock()
		return nil
	}
	cmd := c.ScriptLoad(ctx, s.src)
	if err := cmd.Err(); err != nil {
		s.mu.Unlock()
		return err
	}
	s.hash = cmd.Val()
	s.mu.Unlock()
	return nil
}

func (s *Script) EvalSha(ctx context.Context, c Scripter, keys []string, args ...interface{}) *Cmd {
	// Default behavior: use client-side SHA-1 computed in NewScript.
	if !s.serverSHA {
		s.mu.RLock()
		hash := s.hash
		s.mu.RUnlock()
		return c.EvalSha(ctx, hash, keys, args...)
	}

	// Server-side SHA via SCRIPT LOAD + EVALSHA.
	if err := s.ensureHash(ctx, c); err != nil {
		return s.Eval(ctx, c, keys, args...)
	}

	s.mu.RLock()
	hash := s.hash
	s.mu.RUnlock()

	r := c.EvalSha(ctx, hash, keys, args...)
	if HasErrorPrefix(r.Err(), "NOSCRIPT") {
		// Script cache was flushed; reload and retry once.
		if err := s.ensureHash(ctx, c); err != nil {
			return s.Eval(ctx, c, keys, args...)
		}
		s.mu.RLock()
		hash = s.hash
		s.mu.RUnlock()
		return c.EvalSha(ctx, hash, keys, args...)
	}

	return r
}

func (s *Script) EvalShaRO(ctx context.Context, c Scripter, keys []string, args ...interface{}) *Cmd {
	if !s.serverSHA {
		s.mu.RLock()
		hash := s.hash
		s.mu.RUnlock()
		return c.EvalShaRO(ctx, hash, keys, args...)
	}

	if err := s.ensureHash(ctx, c); err != nil {
		return s.EvalRO(ctx, c, keys, args...)
	}

	s.mu.RLock()
	hash := s.hash
	s.mu.RUnlock()

	r := c.EvalShaRO(ctx, hash, keys, args...)
	if HasErrorPrefix(r.Err(), "NOSCRIPT") {
		if err := s.ensureHash(ctx, c); err != nil {
			return s.EvalRO(ctx, c, keys, args...)
		}
		s.mu.RLock()
		hash = s.hash
		s.mu.RUnlock()
		return c.EvalShaRO(ctx, hash, keys, args...)
	}

	return r
}

// Run optimistically uses EVALSHA to run the script. If script does not exist
// it is retried using EVAL.
func (s *Script) Run(ctx context.Context, c Scripter, keys []string, args ...interface{}) *Cmd {
	r := s.EvalSha(ctx, c, keys, args...)
	if isNoScriptErr(r.Err()) {
		return s.Eval(ctx, c, keys, args...)
	}
	return r
}

// RunRO optimistically uses EVALSHA_RO to run the script. If script does not exist
// it is retried using EVAL_RO.
func (s *Script) RunRO(ctx context.Context, c Scripter, keys []string, args ...interface{}) *Cmd {
	r := s.EvalShaRO(ctx, c, keys, args...)
	if isNoScriptErr(r.Err()) {
		return s.EvalRO(ctx, c, keys, args...)
	}
	return r
}

// isNoScriptErr reports whether err means "this digest is not cached", whether
// it arrived already normalized to ErrNoScript or as the server's raw NOSCRIPT
// error. Both are accepted because the Eval wrappers only normalize when the
// result is readable without blocking — on the deferred autopipeline face the
// raw error reaches here untouched (see cmdable.eval).
func isNoScriptErr(err error) bool {
	if err == nil {
		return false
	}
	return errors.Is(err, ErrNoScript) || HasErrorPrefix(err, "NOSCRIPT")
}

##### 二、一个必须先对齐的前提：go-redis 里没有“锁”

你说“倾向于用 go-redis 的源码，因为他最常用”——**go-redis 本身不实现分布式锁**，全仓库搜不到 `Lock`。它提供的是三块**原语**：

| go-redis 里的源码 | 作用 | 对应锁的哪一步 |
|---|---|---|
| `SetNX` / `SetArgs` / `MSetNX` | 发一条 `SET ... NX PX` | 加锁 |
| `Script.Run`（EVALSHA → EVAL 回退） | 执行 Lua | 解锁 / 续期 |
| `BoolCmd.readReply` | 把 Redis 的 nil 解析成 `false` | “锁被占”不是错误 |

**轮询、退避、续期、重试，go-redis 一行都没有**——这些在 `bsm/redislock` 里（它就是你 `go.mod` 里最可能出现的那个 `redislock`，`go-redis` 官方 README 推荐）。所以“用 go-redis 源码讲主动轮询”的准确玩法是：**用 go-redis 的三个原语手写一个轮询**，再把 `redislock` 当“标准答案”对照。面试让你手写锁的时候，你写的就是这个。

##### 四、板块 1：原子性（go-redis 源码逐行）

###### 4.1 `SetNX` 到底发了什么命令

【逐行理解】`github.com/redis/go-redis/master/string_commands.go`：

```go
func (c cmdable) SetNX(ctx context.Context, key string, value interface{}, expiration time.Duration) *BoolCmd {
	var cmd *BoolCmd
	switch expiration {
	case 0:
		cmd = NewBoolCmd(ctx, "set", key, value, "nx")          // ← 新版；老版是 "setnx"
	case KeepTTL:
		cmd = NewBoolCmd(ctx, "set", key, value, "keepttl", "nx")
	default:
		if usePrecise(expiration) {
			cmd = NewBoolCmd(ctx, "set", key, value, "px", formatMs(ctx, expiration), "nx")
		} else {
			cmd = NewBoolCmd(ctx, "set", key, value, "ex", formatSec(ctx, expiration), "nx")
		}
	}

	_ = c(ctx, cmd)
	return cmd
}
```

四个知识点：

1. **`expiration > 0` 时只发一条 `SET key value PX <ms> NX`**。“判断 key 不存在 + 写入 + 设置过期”三合一，**原子性来自 Redis 单线程串行执行命令**，不是客户端。你博客“问题1”的答案就在这一个 `switch` 分支里。
2. **`expiration == 0` 时没有 TTL**——不管新旧版都一样。这就是你博客“问题2（宕机死锁）”的复现路径：`SetNX(ctx, key, val, 0)` 一行就能写出一个永久死锁。**代码本身不拦你**，得靠纪律/封装。
3. **`usePrecise`**：判断 `expiration` 是否是整秒。不是整秒就发 `PX`（毫秒），是整秒才发 `EX`（秒）。锁场景基本都走 `PX`。
4. **`KeepTTL`**：`SET key value KEEPTTL NX`——保留原 TTL。给锁做“续期但不改 TTL”时有用，属于冷门分支。

###### 4.2 为什么 `SetNX` 返回 `false` 时不报错（源码级证据）

【逐行理解】`command.go` 的 `BoolCmd.readReply`：

```go
func (cmd *BoolCmd) readReply(rd *proto.Reader) (err error) {
	cmd.val, err = rd.ReadBool()

	// `SET key value NX` returns nil when key already exists. But
	// `SETNX key value` returns bool (0/1). So convert nil to bool.
	if err == Nil {
		cmd.val = false
		err = nil
	}
	return err
}
```

Redis 在 `SET ... NX` 因 key 已存在而失败时，回的是 **nil（RESP2 的 bulk nil）**，理论上 go-redis 会把它当 `redis.Nil` 错误。`BoolCmd` 专门把它**吞成 `false`**，注释直接写了原因。

**这个细节在锁里是好事**：`ok, err := rdb.SetNX(...).Result()` 里，`err != nil` 只代表网络/服务端问题，`ok == false` 才是“锁被占”，轮询分支干干净净。

**但它埋了一个反向陷阱**：如果你不用 `SetNX` 而用 `SetArgs`（`StatusCmd`）或自己 `Eval`，key 已存在时拿到的是 **`redis.Nil` 错误**，不是 false。很多“手写锁”的 bug 就出在这：把 `redis.Nil` 当异常直接 return，锁永远抢不到。

###### 4.3 新版还给了什么（这是你“想讲新版本”的素材）

【理解】master 的 `StringCmdable` 接口里有一批 Redis 8.4+ 的实验性命令：

```go
MSetNX(ctx, values ...interface{}) *BoolCmd
SetIFEQ(ctx, key string, value, matchValue interface{}, expiration time.Duration) *StatusCmd
SetIFNE(...)
SetIFDEQ(ctx, key string, value interface{}, matchDigest uint64, expiration time.Duration) *StatusCmd
DelExArgs(ctx, key string, a DelExArgs) *IntCmd
Digest(ctx, key string) *DigestCmd
```

三个和锁直接相关的：

- **`MSetNX`**：一次给多个 key 原子占坑。`redislock` master 版的 `obtain.lua` 里那句 `redis.call("msetnx", unpack(setArgs))` 就是它——**多资源锁（A 转账要同时锁两个账户）必须靠它，`SET NX` 只能锁一个 key**。
- **`SetIFEQ` / `DelExArgs(IFEQ)`**：Redis 8.4 的 compare-and-set / compare-and-delete。理论上**解锁可以不用 Lua 了**：`DELEX key IFEQ <token>` 一条命令完成“比对 + 删除”。go-redis 注释里明确标了 `NOTE ... still experimental`。
- **`Digest` + `SetIFDEQ`**：用 xxh3 摘要做 CAS，值大时省带宽。

【略读】但**现在别在生产赌这些**：8.4+、实验性、云厂商未必支持。知道“未来解锁可能不需要 Lua”就够了——面试说出来是加分，写进项目是风险。

##### 五、板块 2：主动轮询（go-redis 原语手写版）

go-redis 没有现成轮询，手写出来长这样（这版**故意写成“朴素版”**，方便对照 `redislock`）：

```go
var ErrNotObtained = errors.New("lock: not obtained")

func Lock(ctx context.Context, rdb *redis.Client, key, token string,
	ttl, maxWait time.Duration) error {

	ctx, cancel := context.WithTimeout(ctx, maxWait) // ① 总等待上限
	defer cancel()

	backoff := 4 * time.Millisecond
	for {
		ok, err := rdb.SetNX(ctx, key, token, ttl).Result() // ② 一条 SET NX PX
		if err != nil {
			return err                                  // ③ 通信错误：直接失败
		}
		if ok {
			return nil                                  // ④ 拿到锁
		}

		select {
		case <-ctx.Done():                              // ⑤ 取消优先
			return ErrNotObtained
		case <-time.After(backoff):                     // ⑥ 等下一轮
		}

		backoff *= 2                                    // ⑦ 指数退避
		if backoff > 500*time.Millisecond {
			backoff = 500 * time.Millisecond
		}
	}
}
```

逐点（对照 `redislock` v0.8 的 `Obtain`）：

- **① 总等待上限**：等价于 `redislock` 里“ctx 没 deadline 就用 ttl 兜底”那 5 行。**没有它，一个 `context.Background()` 就能让 goroutine 永远轮询。**
- **② 为什么是 `SetNX` 而不是 `Set NX` 手动拼**：`SetNX` 已经把 `PX` 和 `NX` 拼好了，你拼错顺序（`NX PX` vs `PX NX`）Redis 会直接报语法错误——这是新手第一坑。
- **③ 通信错误不重试**：和 `redislock` v0.8 一致（`if err != nil { return nil, err }`）。master 改成了“瞬时错误可重试”，但那是更复杂的设计，**初学阶段“错就返回”是对的**。
- **⑤⑥⑦ 与 `redislock` 结构完全同构**：`timer + select ctx.Done / timer.C`、`backoff < 1 就停`、指数翻倍。
- **⑥ 这里我用 `time.After` 而不是 `Timer`，是个真实的坑**：循环里每次都 `time.After(backoff)` 会**新建一个 Timer**，前面的没触发就扔在那，高频轮询下 GC 压力明显；而且 ctx 取消后 timer 还活着。`redislock` 用 `time.NewTimer` + `Reset` 复用，就是为了这个。**面试被问“轮询怎么实现”，你主动说“用 Timer.Reset 复用而不是 time.After”就是实现层加分。**
- **缺 jitter**：所有等待者同一个退避节奏，会在同一毫秒一起打 Redis（惊群）。加一行 `backoff += time.Duration(rand.Int63n(int64(backoff/2)))` 就散开了。

对比 `redislock` v0.8 的 `Obtain`：除了多了 `randomToken()`、`Options`、`Lock` 结构体，**循环体逐行等价**。这也回答了你上一个问题：**轮询是“模式”，不是某条命令**；从 `SETNX` → `SET NX PX` → `MSETNX + Lua`，命令换了三代，循环骨架没变一行。

###### 一个必须讲清楚的运行时细节：`Script.Run` 不是一条命令

【理解】`redislock` 的 `Release` 是 `luaRelease.Run(...)`，而 `go-redis/script.go` 的 `Run` 是这样：

```go
func (s *Script) Run(ctx context.Context, c Scripter, keys []string, args ...interface{}) *Cmd {
	r := s.EvalSha(ctx, c, keys, args...)      // 先试 EVALSHA（只传 40 字节 sha1）
	if isNoScriptErr(r.Err()) {                // 服务端脚本缓存里没有
		return s.Eval(ctx, c, keys, args...)   // 回退 EVAL（传整段脚本）
	}
	return r
}
```

含义：

- **第一次解锁是 2 个 RTT**（`EVALSHA` 收到 `NOSCRIPT` → `EVAL`）。之后服务端缓存了这个 sha1，就一直是 1 个 RTT。
- 所以“Lua 脚本很轻”是有前提的：**脚本缓存命中**。Redis 重启 / `SCRIPT FLUSH` / 故障切主后会重新走一遍 2 RTT。
- 想预热可以用 `Script.Load`（`SCRIPT LOAD`）——`redislock` 没这么做，因为它接受第一次的额外开销。
- `isNoScriptErr` 同时认 `NEVER`（`redis.ErrNoScript`）和原始 `NOSCRIPT` 前缀，因为自动 pipelining 路径下错误没被规整过——这是 go-redis 内部的角落，**知道有“两种错误形态”就够了，不用记**。

这个点和你下一个板块（解锁 Lua）是连着的：**Lua 的原子性靠 Redis 单线程执行脚本，但“调用一次 Lua”在网络层可能是两次往返**。这两件事不矛盾，很多人混在一起说错。
